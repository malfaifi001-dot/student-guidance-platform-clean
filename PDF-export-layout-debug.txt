function BlockShell({
  block,
  tone = "slate",
  children,
}: {
  block: ReportTemplateBlock;
  tone?: "slate" | "emerald";
  children: ReactNode;
}) {
  const settings = block.settings || {};
  const showTitle = settings.showTitle !== false;
  const style = settings.style || "card";

  if (style === "plain") {
    return (
      <section>
        {showTitle ? (
          <h4 className="mb-3 text-base font-black text-slate-900">
            {block.title}
          </h4>
        ) : null}

        {children}
      </section>
    );
  }

  if (style === "highlight") {
    return (
      <section className="rounded-3xl border border-amber-100 bg-amber-50 p-5">
        {showTitle ? (
          <h4 className="mb-3 text-base font-black text-amber-950">
            {block.title}
          </h4>
        ) : null}

        {children}
      </section>
    );
  }

  return (
    <section
      className={[
        "rounded-3xl border p-5",
        tone === "emerald"
          ? "border-emerald-100 bg-emerald-50"
          : "border-slate-200 bg-slate-50",
      ].join(" ")}
    >
      {showTitle ? (
        <h4
          className={[
            "mb-3 text-base font-black",
            tone === "emerald" ? "text-emerald-950" : "text-slate-900",
          ].join(" ")}
        >
          {block.title}
        </h4>
      ) : null}

      {children}
    </section>
  );
}
