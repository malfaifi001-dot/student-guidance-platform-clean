   42: export function mergeAdjacentCommitteeItemsTableBlocks(
   43:   blocks: LogicalReportBlock[],
   44: ): LogicalReportBlock[] {
   45:   return blocks.reduce<LogicalReportBlock[]>((merged, block) => {
   46:     const previous = merged[merged.length - 1];

   48:       previous &&
   49:       isCommitteeItemsTable(previous) &&
   50:       isCommitteeItemsTable(block) &&
   51:       text(previous.sourceBlockId) !== "" &&
   52:       text(previous.sourceBlockId) === text(block.sourceBlockId);
   53: 
   54:     if (!sameCommitteeTable) {
   55:       merged.push(block);
   56:       return merged;
   57:     }

   60:       ...previous,
   61:       rows: [

   91: 
   92:   if (!columns.length || !rows.length) {
   93:     return [block];
   94:   }
   95: 
   96:   const sourceBlockId = text(block.sourceBlockId) || text(block.id) || "structured-table";
   97:   const settings = block.tableSettings || {};

  100:     id: text(block.sourceTableId) || sourceBlockId,
