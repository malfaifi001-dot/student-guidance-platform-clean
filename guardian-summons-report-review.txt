type CaseValueRow = WorkflowValueLike & {
  id: string;
  fieldKey: string;
  value: string | null;
  jsonValue: unknown;
};

function cleanText(value: unknown) {
  const text = String(repairPotentialUtf8Mojibake(value) ?? "").trim();

  return text && text !== "null" && text !== "undefined" ? text : "";
}

function normalizeText(value: unknown) {
  return cleanText(value)
    .toLowerCase()
    .replace(/[أإآ]/g, "ا")
    .replace(/ى/g, "ي")
    .replace(/ة/g, "ه")
    .replace(/[_\s-]+/g, "")
    .trim();
}

function getRawStoredValue(row: CaseValueRow) {
  if (row.jsonValue !== null && row.jsonValue !== undefined) {
    return row.jsonValue;
  }

  const text = cleanText(row.value);

  if (!text) return "";

  if (
    (text.startsWith("[") && text.endsWith("]")) ||
    (text.startsWith("{") && text.endsWith("}"))
  ) {
    try {
      return JSON.parse(text);
    } catch {
      return text;
    }
  }

  return text;
}

function asRecord(value: unknown): Record<string, unknown> | null {
  return value && typeof value === "object" && !Array.isArray(value)
    ? (value as Record<string, unknown>)
    : null;
}

function fieldSearchText(row: CaseValueRow) {
  return normalizeText(
    [
      row.fieldKey,
      row.field?.key,
      row.field?.label,
      row.field?.type,
    ].join(" "),
  );
}

function findValueRow(rows: CaseValueRow[], aliases: string[]) {
  const normalizedAliases = aliases.map(normalizeText).filter(Boolean);

  return (
    rows.find((row) => {
      const text = fieldSearchText(row);
      return normalizedAliases.some((alias) => text.includes(alias));
    }) || null
  );
}

function findDisplayValue(rows: CaseValueRow[], aliases: string[]) {
  const row = findValueRow(rows, aliases);

  return row ? cleanText(formatWorkflowDisplayValue(row, rows)) : "";
}

function getObjectText(
  value: Record<string, unknown> | null | undefined,
  keys: string[],
) {
  if (!value) return "";

  for (const key of keys) {
    const text = cleanText(value[key]);
    if (text) return text;
  }

  return "";
}

function extractSelectedValues(value: unknown): string[] {
  if (value === null || value === undefined || value === "") {
    return [];
  }

  if (typeof value === "string" || typeof value === "number") {
    return [String(value).trim()].filter(Boolean);
  }

  if (Array.isArray(value)) {
    return value.flatMap((item) => extractSelectedValues(item));
  }

  if (typeof value === "object") {
    const record = value as Record<string, unknown>;
    return [
      ...extractSelectedValues(record.value),
      ...extractSelectedValues(record.label),
      ...extractSelectedValues(record.name),
      ...extractSelectedValues(record.id),
      ...extractSelectedValues(record.key),
    ];
  }

  return [];
}

function buildReasonItems(
  rows: CaseValueRow[],
  aliases: string[],
): GuardianSummonsReasonItem[] {
  const row = findValueRow(rows, aliases);

  if (!row) return [];

  const rawSelectedValues = extractSelectedValues(getRawStoredValue(row));
  const selectedSet = new Set(rawSelectedValues.map(normalizeText));
  const options = row.field?.options || [];

  if (options.length) {
    return options.map((option, index) => {
      const optionValue = cleanText(option.value);
      const optionLabel = cleanText(option.label);
      const selected =
        selectedSet.has(normalizeText(optionValue)) ||
        selectedSet.has(normalizeText(optionLabel));

      return {
        id: optionValue || optionLabel || `${row.id}-${index}`,
        label: optionLabel || optionValue,
        selected,
      };
    });
  }

  return rawSelectedValues.map((value, index) => ({
    id: `${row.id}-${index}`,
    label: cleanText(stringifyWorkflowRawValue(value)),
    selected: true,
  }));
}

function getHijriYear(value: Date | string | null | undefined) {
  const date = value ? new Date(value) : new Date();

  try {
    const parts = new Intl.DateTimeFormat("ar-SA-u-ca-islamic", {
      year: "numeric",
    }).formatToParts(date);

    return parts.find((part) => part.type === "year")?.value || "";
  } catch {
    return "";
  }
}

async function loadGuardianSummonsCase(
  caseId: string,
  context: {
    isAdmin: boolean;
    schoolAccountId: string | null;
  },
) {
  const caseEntry = await prisma.caseEntry.findFirst({
    where: {
      id: caseId,
      service: {
        slug: "guardian-summons",
      },
      ...(context.isAdmin
        ? {}
        : { schoolAccountId: context.schoolAccountId || "__missing__" }),
    },
    include: {
      schoolAccount: {
        include: {
          profile: true,
        },
      },
      service: true,
      workflow: true,
      student: {
        include: {
          guardian: true,
        },
      },
      createdBy: {
        select: {
          name: true,
          officialName: true,
          jobTitle: true,
        },
      },
      values: {
        include: {
          field: {
            include: {
              options: {
                orderBy: {
                  order: "asc",
                },
              },
            },
          },
        },
        orderBy: {
          createdAt: "asc",
        },
      },
    },
  });

  if (!caseEntry) {
    notFound();
  }

  return caseEntry;
}

export default async function GuardianSummonsReportPage({
  params,
  searchParams,
}: PageProps) {
  const [{ caseId }, query, context] = await Promise.all([
    params,
    searchParams ?? Promise.resolve({} as { print?: string }),
    requireDashboardPageContext(),
  ]);

  const caseEntry = await loadGuardianSummonsCase(caseId, context);
  const currentUser = await prisma.user.findUnique({
    where: {
      id: context.user.id,
    },
    select: {
      name: true,
      officialName: true,
      jobTitle: true,
    },
  });
  const values = caseEntry.values as CaseValueRow[];
  const data = buildGuardianSummonsTemplateData({
    caseEntry,
    values,
    currentUser,
  });

  return (
    <main
      dir="rtl"
      className="min-h-screen bg-slate-100 py-6 print:bg-white print:py-0"
    >
      <GuardianSummonsPrintController shouldPrint={query?.print === "1"} />
      <GuardianSummonsTemplate data={data} />
    </main>
  );
}
