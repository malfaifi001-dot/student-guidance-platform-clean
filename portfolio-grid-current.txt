import type { PortfolioReportField } from "@/lib/portfolio/portfolio-report-content";

export type PortfolioFieldKind = "short" | "medium" | "long" | "list";

export type PortfolioFieldPlacement = {
  field: PortfolioReportField;
  fieldKey: string;
  sourceIndex: number;
  columnStart: number;
  columnSpan: number;
  row: number;
  semanticKind: PortfolioFieldKind;
  measuredHeightPx?: number;
  renderedHeightPx?: number;
  stretchToBandHeight: boolean;
  internalLayout: PortfolioFieldInternalLayout;
};

export type PortfolioFieldInternalLayout = {
  valueColumns: 1 | 2 | 3;
  valuePlacement: "grid" | "list";
};

export type PortfolioFieldBand = {
  id: string;
  columnCount: number;
  items: PortfolioFieldPlacement[];
  heightPx?: number;
};

function normalizeText(value: unknown) {
  return String(value || "")
    .replace(/\s+/g, " ")
    .trim();
}

function getValueItems(field: PortfolioReportField) {
  if (Array.isArray(field.value)) {
    return field.value.map(normalizeText).filter(Boolean);
  }

  const value = normalizeText(field.value);
  return value ? [value] : [];
}

function getPortfolioFieldMetrics(field: PortfolioReportField) {
  const label = normalizeText(field.label);
  const isListValue = Array.isArray(field.value);
  const values = getValueItems(field);
  const valueCharacters = values.reduce((sum, value) => sum + value.length, 0);
  const longestValue = values.reduce(
    (longest, value) => Math.max(longest, value.length),
    0,
  );

  return {
    labelCharacters: label.length,
    valueCharacters,
    longestValue,
    totalCharacters: label.length + valueCharacters,
    valuesCount: values.length,
    isListValue,
  };
}

export function getPortfolioFieldInternalLayout(
  field: PortfolioReportField,
  outerColumnCount: 1 | 2 | 3 | 4 | 5 | 6 = 4,
): PortfolioFieldInternalLayout {
  if (!Array.isArray(field.value)) {
    return { valueColumns: 1, valuePlacement: "list" };
  }

  const values = getValueItems(field);
  const longest = values.reduce(
    (max, value) => Math.max(max, value.length),
    0,
  );
  const total = values.reduce(
    (sum, value) => sum + value.length,
    0,
  );

  if (!values.length) {
    return { valueColumns: 1, valuePlacement: "list" };
  }

  /*
   * Multi-value layout is independent from the OUTER grid count.
   *
   * Example:
   * الصفوف = 6 values
   * المهارات = 5 values
   *
   * Both may be half-width outer cards while each card internally
   * renders its own values using two columns.
   *
   * Never force a vertical list merely because the outer candidate
   * dropped from 4 columns to 3 or 2.
   */
  const canUseInternalGrid =
    values.length >= 4 &&
    values.length <= 12 &&
    longest <= 42 &&
    total <= 300;

  if (!canUseInternalGrid) {
    return { valueColumns: 1, valuePlacement: "list" };
  }

  const canUseThreeColumns =
    outerColumnCount >= 5 &&
    values.length >= 6 &&
    longest <= 24 &&
    total <= 190;

  if (canUseThreeColumns) {
    return { valueColumns: 3, valuePlacement: "grid" };
  }

  return {
    valueColumns: 2,
    valuePlacement: "grid",
  };
}

/** Classifies from rendered content, rather than from field names. */
export function classifyPortfolioField(
  field: PortfolioReportField,
): PortfolioFieldKind {
  const metrics = getPortfolioFieldMetrics(field);

  if (
    !metrics.isListValue &&
    metrics.valuesCount <= 1 &&
    metrics.longestValue <= 36 &&
    metrics.labelCharacters <= 36 &&
    metrics.totalCharacters <= 68
  ) {
    return "short";
  }

  if (
    metrics.isListValue &&
    metrics.valuesCount <= 3 &&
    metrics.longestValue <= 52 &&
    metrics.valueCharacters <= 115
  ) {
    return "medium";
  }

  if (
    metrics.valuesCount <= 1 &&
    metrics.longestValue <= 125 &&
    metrics.totalCharacters <= 165
  ) {
    return "medium";
  }

  if (metrics.valuesCount > 1) return "list";
  return "long";
}

/** Returns a readable semantic span for adaptive candidate column counts. */
export function getPortfolioFieldSpanForColumns(
  field: PortfolioReportField,
  columnCount: 1 | 2 | 3 | 4 | 5 | 6,
) {
  const kind = classifyPortfolioField(field);

  if (kind === "list") {
    const metrics = getPortfolioFieldMetrics(field);

    const compactList =
      metrics.valuesCount >= 4 &&
      metrics.valuesCount <= 12 &&
      metrics.longestValue <= 42 &&
      metrics.valueCharacters <= 300;

    if (compactList) {
      /*
       * Prefer approximately half of the outer grid so two compatible
       * list fields can sit beside each other.
       *
       * 6 cols -> 3 + 3
       * 5 cols -> 2 + 3 after band balancing
       * 4 cols -> 2 + 2
       * 3 cols -> 2 (only pairs when physically reasonable)
       * 2 cols -> 1 + 1
       */
      if (columnCount === 2) return 1;
      if (columnCount === 3) return 2;

      return Math.max(
        2,
        Math.floor(columnCount / 2),
      );
    }

    return columnCount;
  }

  if (kind === "long") {
    return columnCount;
  }

  if (kind === "medium") {
    const metrics = getPortfolioFieldMetrics(field);

    /*
     * Scalar metadata such as:
     * البرنامج / الفصل / اليوم / الأسبوع / قناة التنفيذ
     * should be allowed to occupy one compact slot.
     *
     * This lets a measured 4/5-column candidate actually render
     * 4 or 5 useful cards instead of wasting two columns per field.
     */
    const compactScalar =
      !metrics.isListValue &&
      metrics.valuesCount <= 1 &&
      metrics.labelCharacters <= 34 &&
      metrics.longestValue <= 34 &&
      metrics.totalCharacters <= 62;

    if (compactScalar) {
      return 1;
    }

    /*
     * Slightly larger scalar values still remain compact on wider grids,
     * but receive extra width on 4-column layouts when required.
     */
    const moderateScalar =
      !metrics.isListValue &&
      metrics.valuesCount <= 1 &&
      metrics.labelCharacters <= 44 &&
      metrics.longestValue <= 52 &&
      metrics.totalCharacters <= 92;

    if (moderateScalar && columnCount >= 5) {
      return 1;
    }

    if (columnCount >= 4) {
      return 2;
    }

    return 1;
  }

  return 1;
}

/**
 * Creates a deterministic bounded dense placement plan. It preserves source
 * order except for a two-item lookahead used to fill a currently available
 * slot; wide fields are never advanced over another wide field.
 */
export function getAdaptivePortfolioFieldPlacements(
  fields: PortfolioReportField[],
  columnCount: 1 | 2 | 3 | 4 | 5 | 6,
): PortfolioFieldPlacement[] {
  const placements: PortfolioFieldPlacement[] = [];
  const occupancy: number[][] = [];
  const pending = fields.map((field, sourceIndex) => ({ field, sourceIndex }));
  while (pending.length) {
    const current = pending.shift()!;
    const span = getPortfolioFieldSpanForColumns(current.field, columnCount);
    const row = occupancy.length ? occupancy.length - 1 : 0;
    const used = occupancy[row]?.reduce((sum, item) => sum + item, 0) || 0;
    if (used + span > columnCount && used < columnCount && span < columnCount) {
      const lookaheadIndex = pending.slice(0, 3).findIndex(({ field }) => {
        const candidateSpan = getPortfolioFieldSpanForColumns(field, columnCount);
        return candidateSpan < columnCount && used + candidateSpan <= columnCount;
      });
      if (lookaheadIndex >= 0) {
        const compatible = pending.splice(lookaheadIndex, 1)[0];
        const compatibleSpan = getPortfolioFieldSpanForColumns(compatible.field, columnCount);
          placements.push({ field: compatible.field, fieldKey: compatible.field.key, sourceIndex: compatible.sourceIndex, columnStart: used + 1, columnSpan: compatibleSpan, row, semanticKind: classifyPortfolioField(compatible.field), stretchToBandHeight: true, internalLayout: getPortfolioFieldInternalLayout(compatible.field, columnCount) });
        occupancy[row] = [...(occupancy[row] || []), compatibleSpan];
        pending.unshift(current);
        continue;
      }
    }
    const nextRow = used + span > columnCount || !occupancy[row]?.length ? row : row;
    if (!occupancy[nextRow] || (occupancy[nextRow].reduce((sum, item) => sum + item, 0) + span > columnCount)) occupancy.push([]);
    const targetRow = occupancy.length - 1;
    const targetUsed = occupancy[targetRow].reduce((sum, item) => sum + item, 0);
    placements.push({ field: current.field, fieldKey: current.field.key, sourceIndex: current.sourceIndex, columnStart: targetUsed + 1, columnSpan: span, row: targetRow, semanticKind: classifyPortfolioField(current.field), stretchToBandHeight: true, internalLayout: getPortfolioFieldInternalLayout(current.field, columnCount) });
    occupancy[targetRow].push(span);
  }
  return placements;
}

export function getPortfolioFieldBandPlan(fields: PortfolioReportField[], columnCount: 1 | 2 | 3 | 4 | 5 | 6 = 4, measuredHeights?: Record<string, number>): PortfolioFieldBand[] {
  const placements = getAdaptivePortfolioFieldPlacements(fields, columnCount).map((placement) => {
    const measuredHeightPx = measuredHeights?.[placement.fieldKey];
    return { ...placement, measuredHeightPx, renderedHeightPx: measuredHeightPx, stretchToBandHeight: true };
  });
  const bands = new Map<number, PortfolioFieldPlacement[]>();
  for (const placement of placements) bands.set(placement.row, [...(bands.get(placement.row) || []), placement]);
  return Array.from(bands.entries()).sort(([a], [b]) => a - b).map(([row, items]) => {
    const heightPx = items.reduce((max, item) => Math.max(max, item.measuredHeightPx || 0), 0) || undefined;
    return { id: `portfolio-field-band-${row}`, columnCount, heightPx, items: items.map((item) => ({ ...item, renderedHeightPx: heightPx || item.measuredHeightPx, stretchToBandHeight: items.length > 1 })) };
  });
}

function getPortfolioReportPageModel(payload: unknown): PortfolioReportPageModel | undefined {
  if (!payload || typeof payload !== "object" || !("page" in payload)) return undefined;
  const model = (payload as { page?: unknown }).page;
  return model && typeof model === "object" ? model as PortfolioReportPageModel : undefined;
}

export function applyPortfolioSmartCandidate(document: PortfolioPhysicalDocument, candidate: PortfolioSmartCandidate, frozen = false, fieldHeights?: Record<string, number>): PortfolioPhysicalDocument {
  let pages: PortfolioPhysicalPage[] = document.pages.map((page) => {
    const model = getPortfolioReportPageModel(page.payload);
    const details = model?.sections.find((section): section is Extract<PortfolioReportPageModel["sections"][number], { kind: "details" }> => section.kind === "details");
    const fieldBands = details ? getPortfolioFieldBandPlan(details.fields, candidate.columnCount, fieldHeights) : undefined;
    if (details?.fields.length && !fieldBands?.length && process.env.NODE_ENV !== "production") {
      throw new Error(`Portfolio invariant failed: report page ${page.id} has detail fields but no field bands.`);
    }
    const rawPayload = page.payload && typeof page.payload === "object" ? page.payload as Record<string, unknown> : {};
    return {
      ...page,
      density: candidate.id as never,
      candidateId: candidate.id,
      fieldLayout: candidate.fieldLayout,
      fieldColumnCount: candidate.columnCount,
      frozen,
      fieldBands,
      payload: model ? { ...rawPayload, page: { ...model, fieldLayout: candidate.fieldLayout, fieldColumnCount: candidate.columnCount, fieldBands } } : page.payload,
    };
  });
  const candidatePageById = new Map(pages.map((page) => [page.id, page]));
  const reportPages: Record<string, PortfolioPhysicalPage[]> = {};
  for (const [reportId, sourcePages] of Object.entries(document.reportPages)) {
    const candidateSourcePages = sourcePages.map((sourcePage) => candidatePageById.get(sourcePage.id) || sourcePage);
    const primary = candidateSourcePages[0];
    if (!primary) continue;
    const allEvidence = candidateSourcePages.flatMap((sourcePage) => {
      const payload = sourcePage.payload;
      if (!payload || typeof payload !== "object" || !("page" in payload)) return [];
      const model = (payload as { page?: PortfolioReportPageModel }).page;
      return model?.sections.flatMap((section) => section.kind === "evidence" ? section.items : []) || [];
    });
    const count = Math.min(candidate.evidenceCount, allEvidence.length) as 0 | 1 | 2;
    const currentModel = getPortfolioReportPageModel(primary.payload);
    const regularSections = (currentModel?.sections || []).filter((section) => section.kind !== "evidence");
    const rebuilt: PortfolioPhysicalPage[] = [{
      ...primary,
      primaryEvidenceCount: count,
      overflowEvidenceCount: Math.max(0, allEvidence.length - count),
      payload: { ...(primary.payload as object), page: { ...currentModel, sections: [...regularSections, ...(count ? [{ kind: "evidence" as const, items: allEvidence.slice(0, count) }] : [])] } },
    }];
    const template = candidateSourcePages[1] || primary;
    for (let index = count; index < allEvidence.length; index += 2) {
      const items = allEvidence.slice(index, index + 2);
      const templateModel = getPortfolioReportPageModel(template.payload);
      rebuilt.push({
        ...template,
        id: `${primary.id}-evidence-${Math.floor(index / 2) + 1}`,
        continuationIndex: rebuilt.length,
        fieldBands: undefined,
        fieldColumnCount: undefined,
        fieldLayout: undefined,
        primaryEvidenceCount: 0,
        overflowEvidenceCount: Math.max(0, allEvidence.length - index - items.length),
        payload: templateModel ? { ...(template.payload as object), page: { ...templateModel, fieldBands: undefined, fieldColumnCount: undefined, fieldLayout: undefined, key: `${primary.id}-evidence-${Math.floor(index / 2) + 1}`, sections: [{ kind: "evidence" as const, items }] } } : template.payload,
      });
    }
    reportPages[reportId] = rebuilt;
    const ids = new Set(sourcePages.map((page) => page.id));
    pages = [...pages.filter((page) => !ids.has(page.id)), ...rebuilt];
  }
  const finalized: PortfolioPhysicalDocument = {
    ...document,
    pages,
    serviceOutputPages: Object.fromEntries(Object.entries(document.serviceOutputPages).map(([key, items]) => [key, items.map((page) => pages.find((candidatePage) => candidatePage.id === page.id) || page)])),
    reportPages: Object.fromEntries(Object.entries(reportPages).map(([key, items]) => [key, items])),
    evidencePages: Object.fromEntries(Object.entries(reportPages).map(([key, items]) => [key, items.filter((page) => page.primaryEvidenceCount === 0 && Boolean((page.payload as { page?: PortfolioReportPageModel } | undefined)?.page?.sections.some((section) => section.kind === "evidence")))])),
  };
  if (process.env.NODE_ENV !== "production") {
    for (const page of finalized.reportPages ? Object.values(finalized.reportPages).flat() : []) {
      const model = getPortfolioReportPageModel(page.payload);
      const details = model?.sections.find((section): section is Extract<PortfolioReportPageModel["sections"][number], { kind: "details" }> => section.kind === "details");
      if (details?.fields.length) {
        const bandItemCount = page.fieldBands?.reduce((count, band) => count + band.items.length, 0) || 0;
        if (!page.fieldBands?.length || bandItemCount !== details.fields.length) {
          throw new Error(`Portfolio invariant failed after finalization: ${page.id} has ${details.fields.length} detail fields but ${bandItemCount} field-band items.`);
        }
      }
    }
  }
  return finalized;
}

function renderValue(value: string | string[], field?: PortfolioReportField, internalLayout?: PortfolioFieldInternalLayout) {
  if (Array.isArray(value)) {
    const columns = internalLayout?.valueColumns || 1;
    return (
      <ul className="atlas-value-list" style={columns > 1 ? { display: "grid", gridTemplateColumns: `repeat(${columns}, minmax(0, 1fr))`, gap: ".35mm 1mm" } : undefined}>
        {value.map((item, index) => (
          <li key={`${item}-${index}`}>{item}</li>
        ))}
      </ul>
    );
  }

  return value || "غير محدد";
}

function AtlasReportPages({ report, physicalDocument, reportId }: { report: PortfolioReportContent; physicalDocument: PortfolioPhysicalDocument; reportId: string }) {
  const plannedPages = getReportPhysicalPages(physicalDocument, reportId);
  const pages = plannedPages;
  const evidenceHeightMm = getPortfolioEvidenceImageHeightMm(report);
  const evidenceColumns =
    getPortfolioEvidencePerPage(report) <= 1
      ? "minmax(0, 1fr)"
      : "repeat(2, minmax(0, 1fr))";

  return (
    <>
      {pages.map((page, pageIndex) => (
        <AtlasPage
          key={page.key}
          className="portfolio-report-page"
          sectionLabel={report.serviceName || "التقرير"}
          indexLabel={String(pageIndex + 1).padStart(2, "0")}
        >
          <header className="atlas-report-title">
            <span>{report.serviceName || report.subtitle || "تقرير"}</span>
            <div><i aria-hidden="true" /><h1>{report.title}</h1></div>
            <small>صفحة {pageIndex + 1} من {pages.length}</small>
          </header>

          <div className="atlas-report-sections">
            {page.sections.map((section, sectionIndex) => {
              if (section.kind === "details") {
                if (!section.fields.length) return null;

                return (
                  <section
                    key={`details-${page.key}-${sectionIndex}`}
                    className="atlas-report-section"
                  >
                    <h2>التفاصيل</h2>
                    <div className="atlas-detail-grid" style={{ gridTemplateColumns: `repeat(${page.fieldColumnCount || 4}, minmax(0, 1fr))` }}>
                      {(page.fieldBands || []).flatMap((band) =>
                        band.items.map(({ field, columnSpan: effectiveSpan, semanticKind: kind, sourceIndex: index, renderedHeightPx, internalLayout }) => {
                        const tone = index % 3 === 0 ? "coral" : index % 2 === 0 ? "teal" : "blue";

                        return (
                          <article
                            key={`${field.key}-${field.label}`}
                            className={[
                              "atlas-detail-card",
                              `atlas-detail-card-${tone}`,
                              effectiveSpan === 4 ? "atlas-detail-card-wide" : "",
                            ].filter(Boolean).join(" ")}
                            data-portfolio-field-kind={kind}
                            data-portfolio-field-key={field.key}
                            data-portfolio-field-band={band.id}
                            data-portfolio-internal-columns={internalLayout.valueColumns}
                            style={{ gridColumn: `span ${effectiveSpan}`, height: "100%", alignSelf: "stretch", ...(renderedHeightPx ? { minHeight: `${renderedHeightPx}px` } : {}) }}
                          >
                            <span className="atlas-detail-index">
                              {String(index + 1).padStart(2, "0")}
                            </span>
                            <div>
                              <small>{field.label}</small>
                              <strong>{renderValue(field.value, field, internalLayout)}</strong>
                            </div>
                          </article>
                        );
                        }),
                      )}
                    </div>
                  </section>
                );
              }

              if (section.kind === "narrative") {
                if (!section.body.trim()) return null;
                return (
                  <section
                    key={`narrative-${page.key}-${sectionIndex}`}
                    className="atlas-report-section"
                  >
                    <h2>وصف التنفيذ</h2>
                    <p className="atlas-narrative">{section.body}</p>
                  </section>
                );
              }

              if (!section.items.length) return null;

              return (
                <section
                  key={`evidence-${page.key}-${sectionIndex}`}
                  className="atlas-report-section"
                >
                  <h2>الشواهد والمرفقات</h2>
                  <div
                    className="atlas-evidence-grid"
                    style={{ gridTemplateColumns: evidenceColumns }}
                  >
                    {section.items.map((item) => {
                      const isImage =
                        item.type === "IMAGE" ||
                        Boolean(item.url && /\.(png|jpe?g|webp|gif|svg)$/i.test(item.url));

                      if (!item.url) return null;

                      return isImage ? (
                        <figure key={item.id} className="atlas-evidence">
                          <img
                            src={item.url}
                            alt={item.title?.trim() || "صورة مرفقة"}
                            style={{
                              height: `${evidenceHeightMm}mm`,
                              objectFit: report.evidenceSettings.fit,
                            }}
                          />
                        </figure>
                      ) : (
                        <a
                          key={item.id}
                          className="atlas-file"
                          href={item.url}
                          target="_blank"
                          rel="noreferrer"
                        >
                          فتح الملف المرفق
                        </a>
                      );
                    })}
                  </div>
                </section>
              );
            })}
          </div>
        </AtlasPage>
      ))}
    </>
  );
}
