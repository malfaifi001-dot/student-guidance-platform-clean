function getPortfolioReportPageModel(payload: unknown): PortfolioReportPageModel | undefined {
  if (!payload || typeof payload !== "object" || !("page" in payload)) return undefined;
  const model = (payload as { page?: unknown }).page;
  return model && typeof model === "object" ? model as PortfolioReportPageModel : undefined;
}

export function applyPortfolioSmartCandidate(document: PortfolioPhysicalDocument, candidate: PortfolioSmartCandidate, frozen = false, fieldHeights?: Record<string, number>): PortfolioPhysicalDocument {
  let pages: PortfolioPhysicalPage[] = document.pages.map((page) => {
    const model = getPortfolioReportPageModel(page.payload);
    const details = model?.sections.find((section): section is Extract<PortfolioReportPageModel["sections"][number], { kind: "details" }> => section.kind === "details");
    const fieldBands = details ? getPortfolioFieldBandPlan(details.fields, candidate.columnCount, fieldHeights) : undefined;
    if (details?.fields.length && !fieldBands?.length && process.env.NODE_ENV !== "production") {
      throw new Error(`Portfolio invariant failed: report page ${page.id} has detail fields but no field bands.`);
    }
    const rawPayload = page.payload && typeof page.payload === "object" ? page.payload as Record<string, unknown> : {};
    return {
      ...page,
      density: candidate.id as never,
      candidateId: candidate.id,
      fieldLayout: candidate.fieldLayout,
      fieldColumnCount: candidate.columnCount,
      frozen,
      fieldBands,
      payload: model ? { ...rawPayload, page: { ...model, fieldLayout: candidate.fieldLayout, fieldColumnCount: candidate.columnCount, fieldBands } } : page.payload,
    };
  });
  const candidatePageById = new Map(pages.map((page) => [page.id, page]));
  const reportPages: Record<string, PortfolioPhysicalPage[]> = {};
  for (const [reportId, sourcePages] of Object.entries(document.reportPages)) {
    const candidateSourcePages = sourcePages.map((sourcePage) => candidatePageById.get(sourcePage.id) || sourcePage);
    const primary = candidateSourcePages[0];
    if (!primary) continue;
    const allEvidence = candidateSourcePages.flatMap((sourcePage) => {
      const payload = sourcePage.payload;
      if (!payload || typeof payload !== "object" || !("page" in payload)) return [];
      const model = (payload as { page?: PortfolioReportPageModel }).page;
      return model?.sections.flatMap((section) => section.kind === "evidence" ? section.items : []) || [];
    });
    const count = Math.min(candidate.evidenceCount, allEvidence.length) as 0 | 1 | 2;
    const currentModel = getPortfolioReportPageModel(primary.payload);
    const regularSections = (currentModel?.sections || []).filter((section) => section.kind !== "evidence");
    const rebuilt: PortfolioPhysicalPage[] = [{
      ...primary,
      primaryEvidenceCount: count,
      overflowEvidenceCount: Math.max(0, allEvidence.length - count),
      payload: { ...(primary.payload as object), page: { ...currentModel, sections: [...regularSections, ...(count ? [{ kind: "evidence" as const, items: allEvidence.slice(0, count) }] : [])] } },
    }];
    const template = candidateSourcePages[1] || primary;
    for (let index = count; index < allEvidence.length; index += 2) {
      const items = allEvidence.slice(index, index + 2);
      const templateModel = getPortfolioReportPageModel(template.payload);
      rebuilt.push({
        ...template,
        id: `${primary.id}-evidence-${Math.floor(index / 2) + 1}`,
        continuationIndex: rebuilt.length,
        fieldBands: undefined,
        fieldColumnCount: undefined,
        fieldLayout: undefined,
        primaryEvidenceCount: 0,
        overflowEvidenceCount: Math.max(0, allEvidence.length - index - items.length),
        payload: templateModel ? { ...(template.payload as object), page: { ...templateModel, fieldBands: undefined, fieldColumnCount: undefined, fieldLayout: undefined, key: `${primary.id}-evidence-${Math.floor(index / 2) + 1}`, sections: [{ kind: "evidence" as const, items }] } } : template.payload,
      });
    }
    reportPages[reportId] = rebuilt;
    const ids = new Set(sourcePages.map((page) => page.id));
    pages = [...pages.filter((page) => !ids.has(page.id)), ...rebuilt];
  }
  const finalized: PortfolioPhysicalDocument = {
    ...document,
    pages,
    serviceOutputPages: Object.fromEntries(Object.entries(document.serviceOutputPages).map(([key, items]) => [key, items.map((page) => pages.find((candidatePage) => candidatePage.id === page.id) || page)])),
    reportPages: Object.fromEntries(Object.entries(reportPages).map(([key, items]) => [key, items])),
    evidencePages: Object.fromEntries(Object.entries(reportPages).map(([key, items]) => [key, items.filter((page) => page.primaryEvidenceCount === 0 && Boolean((page.payload as { page?: PortfolioReportPageModel } | undefined)?.page?.sections.some((section) => section.kind === "evidence")))])),
  };
  if (process.env.NODE_ENV !== "production") {
    for (const page of finalized.reportPages ? Object.values(finalized.reportPages).flat() : []) {
      const model = getPortfolioReportPageModel(page.payload);
      const details = model?.sections.find((section): section is Extract<PortfolioReportPageModel["sections"][number], { kind: "details" }> => section.kind === "details");
      if (details?.fields.length) {
        const bandItemCount = page.fieldBands?.reduce((count, band) => count + band.items.length, 0) || 0;
        if (!page.fieldBands?.length || bandItemCount !== details.fields.length) {
          throw new Error(`Portfolio invariant failed after finalization: ${page.id} has ${details.fields.length} detail fields but ${bandItemCount} field-band items.`);
        }
      }
    }
  }
  return finalized;
}
