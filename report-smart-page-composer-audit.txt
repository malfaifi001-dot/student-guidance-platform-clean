  111:   content: HTMLElement,
  112:   priorityMode: ReportSmartA4PriorityMode,
  113:   profile: ReportSmartA4Profile,
  114: ): ReportSmartA4Measurement {
  115:   const viewportRect = viewport.getBoundingClientRect();
  116:   const contentRect = content.getBoundingClientRect();
  117: 
  118:   const localScale =
  119:     viewport.clientHeight > 0
  120:       ? viewportRect.height / viewport.clientHeight
  121:       : 1;
  122: 
  123:   const safeScale =
  124:     Number.isFinite(localScale) && localScale > 0
  125:       ? localScale
  126:       : 1;
  127: 
  128:   const viewportHeightPx = viewport.clientHeight;
  129:   const footerSafeAreaMm = Math.max(
  130:     0,
  131:     Number(viewport.dataset.reportFooterSafeAreaMm || "0") || 0,
  132:   );
  133:   const footerSafeAreaPx = footerSafeAreaMm * (96 / 25.4);
  134:   const footerBoundaryGapPx =
  135:     REPORT_SMART_A4_FOOTER_BOUNDARY_GAP_MM * (96 / 25.4);
  136:   const physicalPage = viewport.closest<HTMLElement>(".pdf-report-page");
  137:   const pageFooter =
  138:     physicalPage?.querySelector<HTMLElement>("[data-report-page-footer]") ||
  139:     null;
  140:   const fallbackBoundaryBottom =
  141:     viewportRect.bottom - footerSafeAreaPx * safeScale;
  142:   const realFooterBoundaryBottom = pageFooter
  143:     ? pageFooter.getBoundingClientRect().top - footerBoundaryGapPx * safeScale
  144:     : Number.POSITIVE_INFINITY;
  145:   const boundaryBottom = Math.min(
  146:     fallbackBoundaryBottom,
  147:     realFooterBoundaryBottom,
  148:   );
  149:   const footerBoundaryUsed = Boolean(pageFooter);
  150:   const usableViewportHeightPx = Math.max(
  151:     0,
  152:     Math.min(
  153:       viewportHeightPx,
  154:       (boundaryBottom - viewportRect.top) / safeScale,
  155:     ),
  156:   );
  157: 
  158:   const elements = Array.from(
  159:     content.querySelectorAll<HTMLElement>(
  160:       '[data-report-smart-block], [data-report-priority-block="signature"], [data-report-design-signature-block]',
  161:     ),
  162:   );
  163: 
  164:   const blocks = elements.map(
  165:     (element): ReportSmartA4BlockMeasurement => {
  166:       const rect = element.getBoundingClientRect();
  167: 
  168:       const topPx =
  169:         (rect.top - contentRect.top) / safeScale;
  170: 
  171:       const bottomPx =
  172:         (rect.bottom - contentRect.top) / safeScale;
  173: 
  174:       return {
  175:         role: getBlockRole(element),
  176:         kind: String(
  177:           element.dataset.reportSmartBlock || "",
  178:         ),
  179:         heightPx: round(rect.height / safeScale),
  180:         topPx: round(topPx),
  181:         bottomPx: round(bottomPx),
  182:         overflowPx: round(
  183:           Math.max(
  184:             0,
  185:             (rect.bottom - boundaryBottom) / safeScale,
  186:           ),
  187:         ),
  188:       };
  189:     },
  190:   );
  191:   const crossesHardBoundary = elements.some(
  192:     (element) => element.getBoundingClientRect().bottom > boundaryBottom,
  193:   );
  194: 
  195:   const signatureBlocks = blocks.filter(
  196:     (block) => block.role === "signature",
  197:   );
  198: 
  199:   const nonSignatureBlocks = blocks.filter(
  200:     (block) => block.role !== "signature",
  201:   );
  202: 
  203:   const signatureHeightPx = signatureBlocks.reduce(
  204:     (sum, block) => sum + block.heightPx,
  205:     0,
  206:   );
  207: 
  208:   const signatureReservedPx =
  209:     priorityMode === "signature" && signatureBlocks.length
  210:       ? signatureHeightPx + profile.signatureSafetyGapPx
  211:       : 0;
  212: 
  213:   const mainContentBudgetPx = Math.max(
  214:     0,
  215:     usableViewportHeightPx - signatureReservedPx,
  216:   );
  217: 
  218:   const mainContentBottomPx = nonSignatureBlocks.reduce(
  219:     (largest, block) =>
  220:       Math.max(largest, block.bottomPx),
  221:     0,
  222:   );
  223: 
  224:   const mainContentOverflowPx =
  225:     priorityMode === "signature"
  226:       ? Math.max(
  227:           0,
  228:           mainContentBottomPx - mainContentBudgetPx,
  229:         )
  230:       : 0;
  231: 
  232:   const blockOverflowPx = blocks.reduce(
  233:     (largest, block) =>
  234:       Math.max(largest, block.overflowPx),
  235:     0,
  236:   );
  237: 
  238:   const scrollOverflowPx = Math.max(
  239:     0,
  240:     content.scrollHeight - viewportHeightPx,
  241:   );
  242: 
  243:   const boundingOverflowPx = Math.max(
  244:     0,
  245:     (contentRect.bottom - viewportRect.bottom) / safeScale,
  246:   );
  247: 
  248:   const overflowPx = Math.max(
  249:     0,
  250:     blockOverflowPx,
  251:     scrollOverflowPx,
  252:     boundingOverflowPx,
  253:     mainContentOverflowPx,
  254:   );
  255: 
  256:   const roleHeights = {
  257:     fields: sumRoleHeight(blocks, "fields"),
  258:     evidence: sumRoleHeight(blocks, "evidence"),
  259:     narrative: sumRoleHeight(blocks, "narrative"),
  260:     table: sumRoleHeight(blocks, "table"),
  261:     signature: signatureHeightPx,
  262:     title: sumRoleHeight(blocks, "title"),
  263:     general: sumRoleHeight(blocks, "general"),
  264:   };
  265: 
  266:   const dominantRole = (
  267:     Object.entries(roleHeights) as Array<
  268:       [ReportSmartA4BlockRole, number]
  269:     >
  270:   ).reduce(
  271:     (best, current) =>
  272:       current[1] > best[1] ? current : best,
  273:     ["general", 0] as [
  274:       ReportSmartA4BlockRole,
  275:       number,
  276:     ],
  277:   )[0];
  278: 
  279:   return {
  280:     fits:
  281:       !crossesHardBoundary &&
  282:       blockOverflowPx <= 0 &&
  283:       overflowPx <= REPORT_SMART_A4_TOLERANCE_PX,
  284: 
  285:     viewportHeightPx: round(usableViewportHeightPx),
  286:     contentHeightPx: round(content.scrollHeight),
  287: 
  288:     overflowPx: round(overflowPx),
  289: 
  290:     signatureHeightPx: round(signatureHeightPx),
  291:     signatureReservedPx: round(signatureReservedPx),
  292: 
  293:     mainContentBottomPx: round(mainContentBottomPx),
  294:     mainContentBudgetPx: round(mainContentBudgetPx),
  295:     mainContentOverflowPx: round(mainContentOverflowPx),
  296: 
  297:     boundaryBottomPx: round(
  298:       (boundaryBottom - viewportRect.top) / safeScale,
  299:     ),
  300:     footerBoundaryUsed,
  301: 
  302:     fieldHeightPx: round(roleHeights.fields),
  303:     evidenceHeightPx: round(roleHeights.evidence),
  304:     narrativeHeightPx: round(roleHeights.narrative),
  305:     tableHeightPx: round(roleHeights.table),
  306: 
  307:     dominantRole,
  308: 
  309:     blocks,
  310:   };
  311: }
