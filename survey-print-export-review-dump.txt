194: function buildQuestionAnalysis(survey: SurveyForPdf): QuestionAnalysis[] {
195:   const totalResponses = survey.responses.length;
196: 
197:   return survey.questions.map((question) => {
198:     const responseAnswers = survey.responses.map((response) =>
199:       response.answers.find((answer) => answer.questionId === question.id),
200:     );
201: 
202:     const answeredAnswers = responseAnswers.filter((answer) => answerToText(answer).trim());
203:     const numericValues = responseAnswers
204:       .map((answer) => answerToNumber(answer))
205:       .filter((value): value is number => value !== null);
206: 
207:     const optionLabels =
208:       question.type === "YES_NO"
209:         ? ["نعم", "لا"]
210:         : question.options.map((option) => option.label);
211: 
212:     const optionCounts = isChoiceQuestion(question.type)
213:       ? optionLabels.map((label) => {
214:           const count = responseAnswers.filter((answer) => {
215:             const values = answerToText(answer)
216:               .split("،")
217:               .map((item) => item.trim())
218:               .filter(Boolean);
219: 
220:             return values.includes(label);
221:           }).length;
222: 
223:           return {
224:             label,
225:             count,
226:             percentage: answeredAnswers.length ? Math.round((count / answeredAnswers.length) * 100) : 0,
227:           };
228:         })
229:       : [];
230: 
231:     const textSamples =
232:       question.type === "TEXT" || question.type === "TEXTAREA"
233:         ? responseAnswers
234:             .map((answer) => answerToText(answer).trim())
235:             .filter(Boolean)
236:             .slice(0, 2)
237:         : [];
238: 
239:     return {
240:       id: question.id,
241:       label: question.label,
242:       type: question.type,
243:       isRequired: question.isRequired,
244:       answeredCount: answeredAnswers.length,
245:       emptyCount: Math.max(totalResponses - answeredAnswers.length, 0),
246:       answerRate: totalResponses ? Math.round((answeredAnswers.length / totalResponses) * 100) : 0,
247:       average:
248:         isNumericQuestion(question.type) && numericValues.length
249:           ? Number((numericValues.reduce((sum, value) => sum + value, 0) / numericValues.length).toFixed(2))
250:           : null,
251:       min: numericValues.length ? Math.min(...numericValues) : null,
252:       max: numericValues.length ? Math.max(...numericValues) : null,
253:       optionCounts,
254:       textSamples,
255:     };
256:   });
