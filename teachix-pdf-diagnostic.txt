function pdfFileName(title: string) {

const DOWNLOAD_REQUEST_ID_PATTERN = /^[A-Za-z0-9_-]{16,48}$/;

function portfolioDownloadDebug(stage: string, details: Record<string, unknown>) {
  console.info("PORTFOLIO_DOWNLOAD_DEBUG", { stage, ...details });
}

function getDownloadRequestId(request: Request) {
  const value = new URL(request.url).searchParams.get("downloadRequestId") || "";
  return DOWNLOAD_REQUEST_ID_PATTERN.test(value) ? value : "";
}

function responseWithPdf(
  pdf: Uint8Array,
  fileName: string,
  downloadRequestId: string,
  elapsedMs: number,
) {
  portfolioDownloadDebug("response-build-start", { elapsedMs, bytesLength: pdf.byteLength });
  // Cloudflare returns a full Uint8Array view. Reuse its ArrayBuffer instead of
  // slicing a second ~23 MB buffer; retain a safe fallback for non-full views.
  const body = pdf.byteOffset === 0 && pdf.byteLength === pdf.buffer.byteLength
    ? pdf.buffer as ArrayBuffer
    : pdf.slice().buffer as ArrayBuffer;
  const response = new NextResponse(body, {
    headers: {
      "Content-Type": "application/pdf",
      "Content-Disposition": `attachment; filename="portfolio.pdf"; filename*=UTF-8''${encodeURIComponent(fileName)}`,
      "Cache-Control": "private, no-store",
      "Content-Length": String(pdf.byteLength),
    },
  });

  if (downloadRequestId) {
    response.cookies.set({
      name: `teachix_portfolio_download_${downloadRequestId}`,
      value: "1",
      path: "/",
      maxAge: 120,
      sameSite: "lax",
      secure: process.env.NODE_ENV === "production",
      httpOnly: false,
    });
  }

  portfolioDownloadDebug("response-build-finished", {
    elapsedMs,
    bytesLength: pdf.byteLength,
    contentType: "application/pdf",
    contentDispositionPresent: true,
    downloadRequestIdPresent: Boolean(downloadRequestId),
  });
  portfolioDownloadDebug("response-return", { elapsedMs, bytesLength: pdf.byteLength });
  return response;
}

    const previewUrl = `${getRequestOrigin(request)}/portfolio-export-preview/${encodeURIComponent(token)}?pdf=1`;
    console.info("PORTFOLIO_CLOUDFLARE_DEBUG", {
      stage: "export-route",
      previewPath: "/portfolio-export-preview/[token]?pdf=1",
    });
    const pdf = await generatePdfFromUrlWithCloudflare({
      url: previewUrl,
      gotoWaitUntil: "domcontentloaded",
      waitForSelector: '[data-portfolio-pdf-ready="true"]',
      waitForSelectorTimeoutMs: 30_000,
      debugLabel: "portfolio",
    });
    const downloadRequestId = getDownloadRequestId(request);
    const elapsedMs = Date.now() - startedAt;
    portfolioDownloadDebug("route-pdf-ready", {
      elapsedMs,

import { NextResponse } from "next/server";

import {

  completePortfolioSnapshotPdfJob,
  failPortfolioSnapshotPdfJob,
  portfolioSnapshotPdfStorageKey,
  type PortfolioSnapshotPdfClaimOperation,
} from "@/lib/portfolio/portfolio-snapshot-pdf-service";
import { parsePortfolioSnapshotDocument } from "@/lib/portfolio/portfolio-snapshot-types";

function authorized(request: Request) {
  const expected = process.env.PORTFOLIO_SNAPSHOT_PDF_RUNNER_SECRET?.trim();
  const supplied = request.headers.get("x-portfolio-pdf-runner-secret") || request.headers.get("authorization")?.replace(/^Bearer\s+/i, "") || "";
  if (!expected || !supplied) return false;
  const expectedBytes = Buffer.from(expected);
  const suppliedBytes = Buffer.from(supplied);
  return expectedBytes.length === suppliedBytes.length && timingSafeEqual(expectedBytes, suppliedBytes);
}

function isPdf(data: Uint8Array) {
  return data.byteLength > 5 && Buffer.from(data.subarray(0, 5)).toString("ascii") === "%PDF-";
}

function logPortfolioSnapshotPdf(event: string, details: Record<string, unknown>) {
  console.info(`PORTFOLIO_SNAPSHOT_PDF ${JSON.stringify({ event, ...details })}`);
}

const claimOperations = new Set<PortfolioSnapshotPdfClaimOperation>([
  "lock-connect",
  "get-lock",
  "recover-expired",
  "find-first",
  "claim-update",
  "load-job",
]);

function safeClaimErrorDetails(error: unknown) {
  const candidate = error && typeof error === "object" ? error as Record<string, unknown> : {};
  const operation = candidate.claimOperation;
  const errorName = error instanceof Error ? error.name : "UnknownError";
  const errorCode = candidate.code;
  const errno = candidate.errno;
  const sqlState = candidate.sqlState;

  return {
    claimOperation: typeof operation === "string" && claimOperations.has(operation as PortfolioSnapshotPdfClaimOperation)
      ? operation
      : "unknown",
    errorName: /^[A-Za-z][A-Za-z0-9_.-]{0,79}$/.test(errorName) ? errorName : "UnknownError",
    errorCode: typeof errorCode === "string" && /^[A-Za-z0-9_.:-]{1,64}$/.test(errorCode) ? errorCode : null,
    errno: typeof errno === "number" && Number.isInteger(errno) && Math.abs(errno) <= 999999 ? errno : null,
    sqlState: typeof sqlState === "string" && /^[A-Z0-9]{5}$/.test(sqlState) ? sqlState : null,
  };
}

export async function POST(request: Request) {
  if (!authorized(request)) return NextResponse.json({ ok: false }, { status: 401 });

  const runnerStartedAt = Date.now();
  let stage = "claim";
  let claimResult: Awaited<ReturnType<typeof claimNextPortfolioSnapshotPdfJob>>;
  try {
    claimResult = await claimNextPortfolioSnapshotPdfJob();
  } catch (error) {
    logPortfolioSnapshotPdf("claim-failed", {
      stage,
      durationMs: Date.now() - runnerStartedAt,
      ...safeClaimErrorDetails(error),
    });
    return NextResponse.json({ ok: false, claimed: false, reason: "CLAIM_QUERY_FAILED" }, { status: 500 });
  }

  if (claimResult.kind === "NO_ELIGIBLE_JOB") {
    logPortfolioSnapshotPdf("claim-empty", { stage, durationMs: Date.now() - runnerStartedAt });
    return NextResponse.json({ ok: true, claimed: false, reason: "NO_ELIGIBLE_JOB" });
  }
  if (claimResult.kind === "GLOBAL_BUSY") {
    logPortfolioSnapshotPdf("claim-busy", { stage, durationMs: Date.now() - runnerStartedAt });
    return NextResponse.json({ ok: true, claimed: false, reason: "GLOBAL_PDF_BUSY" }, { status: 202 });
  }
  if (claimResult.kind === "CLAIM_CONFLICT") {
    logPortfolioSnapshotPdf("claim-conflict", { stage, durationMs: Date.now() - runnerStartedAt });
    return NextResponse.json({ ok: false, claimed: false, reason: "CLAIM_CONFLICT" }, { status: 409 });
  }

  const job = claimResult.job;
  const releaseGlobalLock = claimResult.releaseGlobalLock;
  logPortfolioSnapshotPdf("claimed", { stage, durationMs: Date.now() - runnerStartedAt });

  const contentHash = job.contentHash;
  const finalKey = portfolioSnapshotPdfStorageKey(job.snapshotId, contentHash);
  const tempKey = `portfolio-pdfs/snapshots/${job.snapshotId}/.${contentHash}.${job.claimToken}.tmp`;
  try {
    stage = "snapshot-load";
    const document = parsePortfolioSnapshotDocument(job.snapshot.snapshotJson);
    stage = "image-preflight";
    await assertPortfolioExportImageSourcesAvailable(document);
    stage = "existing-artifact";
    if (await storageFileExists(storageKeySegments(finalKey))) {
      const existingPdf = await readStorageFile(storageKeySegments(finalKey));
      if (isPdf(existingPdf) && existingPdf.byteLength >= 100) {
        const finalized = await completePortfolioSnapshotPdfJob({
          id: job.id,
          claimToken: job.claimToken!,
          contentHash,
          storageKey: finalKey,
          byteSize: existingPdf.byteLength,
        });
        if (finalized.count !== 1) throw new Error("PDF_JOB_CLAIM_EXPIRED_BEFORE_FINALIZE");
        logPortfolioSnapshotPdf("ready-existing", { stage, durationMs: Date.now() - runnerStartedAt, byteSize: existingPdf.byteLength });
        return NextResponse.json({ ok: true, claimed: true, status: "READY" });
      }
      await deleteStorageFile(storageKeySegments(finalKey));
    }
    stage = "export-token";
    const token = await createPortfolioExportToken(document);
    stage = "public-origin";
    const previewUrl = `${getPortfolioPublicOrigin()}/portfolio-export-preview/${encodeURIComponent(token)}?pdf=1`;
    stage = "cloudflare";
    const pdf = await generatePdfFromUrlWithCloudflare({
      url: previewUrl,
      gotoWaitUntil: "domcontentloaded",
      waitForSelector: '[data-portfolio-pdf-ready="true"]',
      waitForSelectorTimeoutMs: 60_000,
      debugLabel: "portfolio",
    });
    if (!isPdf(pdf) || pdf.byteLength < 100) throw new Error("CLOUDFLARE_RETURNED_INVALID_PDF");

    stage = "storage-write";
    await writeStorageFile(storageKeySegments(tempKey), pdf);
    stage = "storage-publish";
    await renameStorageFile(storageKeySegments(tempKey), storageKeySegments(finalKey));
    stage = "finalize";
    const finalized = await completePortfolioSnapshotPdfJob({
      id: job.id,
      claimToken: job.claimToken!,
      contentHash,
      storageKey: finalKey,
      byteSize: pdf.byteLength,
    });
    if (finalized.count !== 1) throw new Error("PDF_JOB_CLAIM_EXPIRED_BEFORE_FINALIZE");
    logPortfolioSnapshotPdf("ready", { stage, durationMs: Date.now() - runnerStartedAt, byteSize: pdf.byteLength });
    return NextResponse.json({ ok: true, claimed: true, status: "READY" });
  } catch (error) {
    const failure = await failPortfolioSnapshotPdfJob({ id: job.id, claimToken: job.claimToken!, error });
    logPortfolioSnapshotPdf("failed", { stage, durationMs: Date.now() - runnerStartedAt, errorClass: error instanceof Error ? error.constructor.name : "UnknownError", retryable: failure.retryable });
    return NextResponse.json({ ok: false, claimed: true, status: failure.retryable ? "PENDING" : "FAILED" }, { status: 500 });
  } finally {
    await deleteStorageFile(storageKeySegments(tempKey)).catch(() => undefined);
    await releaseGlobalLock().catch(() => undefined);
  }
}

import { NextResponse } from "next/server";

import { getEvidenceMimeType, getEvidenceStoredFileNameFromUrl } from "@/lib/evidence/evidence-file-storage";
import { collectPortfolioExportImageUrls, readPortfolioExportToken } from "@/lib/portfolio/portfolio-export-snapshot";
import { readDurableUpload, isSafeStorageId, isSafeUploadedFileName } from "@/lib/storage/durable-upload-storage";
import { readStorageFile } from "@/lib/storage/storage-provider";

export const runtime = "nodejs";
export const dynamic = "force-dynamic";

const MAX_INPUT_BYTES = 15 * 1024 * 1024;
const MAX_INPUT_PIXELS = 40_000_000;
const MAX_RASTER_EDGE = 2000;
let activeSharpJobs = 0;
const waitingSharpJobs: Array<() => void> = [];

function notFound() {
  return new NextResponse("Not Found", { status: 404, headers: { "Cache-Control": "no-store", "X-Content-Type-Options": "nosniff" } });
}

async function withSharpSlot<T>(work: () => Promise<T>) {
  if (activeSharpJobs >= 2) await new Promise<void>((resolve) => waitingSharpJobs.push(resolve));
  activeSharpJobs += 1;
  try { return await work(); } finally {
    activeSharpJobs -= 1;
    waitingSharpJobs.shift()?.();
  }
}

function sourceDetails(source: string) {
  if (source.startsWith("/uploads/portfolio/")) {
    const parts = source.slice("/uploads/portfolio/".length).split("/");
    if (parts.length !== 2 || !isSafeStorageId(parts[0]) || !isSafeUploadedFileName(parts[1])) return null;
    return { kind: "portfolio" as const, ownerId: parts[0], fileName: parts[1], mimeType: "image/webp" };
  }
  const fileName = getEvidenceStoredFileNameFromUrl(source);
  const mimeType = fileName ? getEvidenceMimeType(fileName) : null;
  if (!fileName || !mimeType?.startsWith("image/")) return null;
  return { kind: "evidence" as const, fileName, mimeType };
}

async function readSource(source: string, details: ReturnType<typeof sourceDetails>) {
  if (!details) return null;
  if (details.kind === "portfolio") return readDurableUpload("portfolio", details.ownerId, details.fileName);
  return (await readStorageFile(["evidence", details.fileName])) || null;
}

async function transformImage(input: Buffer, mimeType: string) {
  const sharpModule = await import("sharp");
  const sharp = sharpModule.default;
  const image = sharp(input, { limitInputPixels: MAX_INPUT_PIXELS, failOn: "warning" });
  const metadata = await image.metadata();
  if (!metadata.width || !metadata.height) throw new Error("IMAGE_DIMENSIONS_UNAVAILABLE");
  if (input.byteLength <= 750_000 && Math.max(metadata.width, metadata.height) <= MAX_RASTER_EDGE) {
    return { body: input, mimeType };
  }

  const pipeline = image.rotate().resize({ width: MAX_RASTER_EDGE, height: MAX_RASTER_EDGE, fit: "inside", withoutEnlargement: true });
  if (mimeType === "image/png" || metadata.hasAlpha) {
    return { body: await pipeline.png({ compressionLevel: 9, adaptiveFiltering: true }).toBuffer(), mimeType: "image/png" };
  }
  if (mimeType === "image/webp") {
    return { body: await pipeline.webp({ quality: 86, effort: 4 }).toBuffer(), mimeType: "image/webp" };
  }
  return { body: await pipeline.jpeg({ quality: 86, mozjpeg: true, progressive: true }).toBuffer(), mimeType: "image/jpeg" };
}

export async function GET(request: Request) {
  const url = new URL(request.url);
  const token = url.searchParams.get("token") || "";
  const source = url.searchParams.get("source") || "";
  const payload = await readPortfolioExportToken(token);
  if (!payload || !source || !collectPortfolioExportImageUrls(payload.document).has(source)) return notFound();

  const details = sourceDetails(source);
  if (!details) return notFound();
  try {
    const original = await readSource(source, details);
    if (!original || original.byteLength > MAX_INPUT_BYTES) return notFound();
    let result: { body: Buffer; mimeType: string };
    try {
      result = await withSharpSlot(() => transformImage(original, details.mimeType));
    } catch (error) {
      console.warn("PORTFOLIO_SNAPSHOT_IMAGE_TRANSFORM_FALLBACK", {
        optimized: false,
        errorName: error instanceof Error ? error.name : "UnknownError",
        errorCode: error && typeof error === "object" && "code" in error && typeof error.code === "string" ? error.code : null,
      });
      result = { body: original, mimeType: details.mimeType };
    }
    return new NextResponse(new Uint8Array(result.body), {
      headers: {
        "Cache-Control": "private, no-store",
        "Content-Length": String(result.body.byteLength),
        "Content-Type": result.mimeType,
        "X-Content-Type-Options": "nosniff",
      },
    });
  } catch {
    return notFound();
  }
}

  width: 0,
  height: 0,
  scale: 1,
  fitScale: 1,
  logicalWidth: 0,
  logicalHeight: 0,

const MIN_ZOOM_FACTOR = 0.75;
const MAX_ZOOM_FACTOR = 2.5;

  const [dimensions, setDimensions] =
    useState<PreviewDimensions>(INITIAL_DIMENSIONS);
  const dimensionsRef = useRef(INITIAL_DIMENSIONS);
  const zoomFactorRef = useRef(1);
  const geometryRef = useRef<PreviewZoomScrollGeometry>({
    logicalWidth: INITIAL_DIMENSIONS.logicalWidth,
    logicalHeight: INITIAL_DIMENSIONS.logicalHeight,
    fitScale: INITIAL_DIMENSIONS.fitScale,
    scale: INITIAL_DIMENSIONS.scale,
  });

  const applyPreviewScale = (
    nextScale: number,
    nextDimensions: PreviewDimensions,
    shouldUpdateState = true,
  ) => {
    dimensionsRef.current = nextDimensions;

    if (scaledFrameRef.current) {
      scaledFrameRef.current.style.width = `${nextDimensions.width}px`;
      scaledFrameRef.current.style.height = `${nextDimensions.height}px`;
    }

    if (stageRef.current) {
      stageRef.current.style.transform = `scale(${nextScale})`;
    }

    if (shouldUpdateState) setDimensions(nextDimensions);
  };

  const fitToViewport = () => {
    const current = dimensionsRef.current;
    zoomFactorRef.current = 1;
    const nextScale = current.fitScale;
    applyPreviewScale(
      nextScale,
      {
        ...current,
        width: current.logicalWidth * nextScale,
        height: current.logicalHeight * nextScale,
        scale: nextScale,
      },
    );

    previewInteraction.resetZoom();
  };

    enabled: true,
    viewportRef,
    stageRef,
    geometryRef,
    zoomFactorRef,
    minZoomFactor: MIN_ZOOM_FACTOR,
    maxZoomFactor: MAX_ZOOM_FACTOR,
    onZoomFactorChange: (nextZoomFactor) => {
      const current = dimensionsRef.current;
      const nextScale = current.fitScale * nextZoomFactor;
      applyPreviewScale(
        nextScale,
        {
          ...current,
          width: current.logicalWidth * nextScale,
          height: current.logicalHeight * nextScale,
          scale: nextScale,
        },
        false,
      );
    },
  });

    const stage = stageRef.current;

    if (!viewport || !stage) return;

    const updatePreviewScale = () => {
      window.cancelAnimationFrame(frameId);
      frameId = window.requestAnimationFrame(() => {
        const screenWidth =
          window.visualViewport?.width ||
          document.documentElement.clientWidth ||
          window.innerWidth;
        const viewportWidth = Math.min(viewport.clientWidth, screenWidth);
        const safeHorizontalGap =
          viewportWidth < 640 ? 24 : viewportWidth < 1180 ? 32 : 40;
        const availableWidth = Math.max(
          0,
          viewportWidth - safeHorizontalGap,
        );
        const logicalWidth = Math.max(stage.scrollWidth, stage.offsetWidth);
        const logicalHeight = Math.max(stage.scrollHeight, stage.offsetHeight);

        if (!logicalWidth || !logicalHeight || !availableWidth) return;

        const fitScale = Math.min(1, availableWidth / logicalWidth);

        }

        setDimensions((current) => {
          const isUnchanged =
            Math.abs(current.width - nextDimensions.width) < 0.5 &&
            Math.abs(current.height - nextDimensions.height) < 0.5 &&
            Math.abs(current.scale - nextDimensions.scale) < 0.001 &&
            Math.abs(current.fitScale - nextDimensions.fitScale) < 0.001;

import type { ReactNode } from "react";
import { useEffect, useRef, useState } from "react";

import { PrintExportPopCard } from "@/components/print-export/print-export-pop-card";
import type { PrintExportModal } from "@/lib/print-export/print-export-types";

const READY_SELECTOR = '[data-portfolio-pdf-ready="true"]';
const FAILURE_TIMEOUT_MS = 60_000;
const COMPLETION_DELAY_MS = 450;

export function PortfolioPreviewLoadingBoundary({ children }: { children: ReactNode }) {
  const pathname = usePathname();
  const [progress, setProgress] = useState(0);
  const [active, setActive] = useState(() => pathname.includes("/portfolio/preview"));
  const [session, setSession] = useState(0);
  const [ready, setReady] = useState(false);
  const [failed, setFailed] = useState(false);
  const [dismissed, setDismissed] = useState(false);

  const runId = useRef(0);
  const readyRef = useRef(false);
  useEffect(() => {
    const start = () => {
      runId.current += 1;
      readyRef.current = false;
      setActive(true); setReady(false); setFailed(false); setDismissed(false); setProgress(0); setSession((current) => current + 1);
    };
    window.addEventListener("portfolio-preview-start", start);
    if (pathname.includes("/portfolio/preview")) start();
    else setActive(false);
    return () => window.removeEventListener("portfolio-preview-start", start);
  }, [pathname]);

  useEffect(() => {
    if (!active) return;
    const currentRun = runId.current;
    const startedAt = performance.now();
    let completionTimer: number | null = null;
    const progressTimer = window.setInterval(() => {
      setProgress((current) => {
        if (current >= 94 || ready) return current;
        const elapsed = performance.now() - startedAt;
        return Math.min(94, Math.max(current, Math.round(94 * (1 - Math.exp(-elapsed / 6500)))));
      });
    }, 120);

    const finish = () => {
      if (currentRun !== runId.current || readyRef.current) return;
      readyRef.current = true;
      setReady(true);
      setProgress(100);
      completionTimer = window.setTimeout(() => setDismissed(true), COMPLETION_DELAY_MS);
    };

    const checkReady = () => {
      if (document.querySelector(READY_SELECTOR)) finish();
    };

    checkReady();
    const observer = new MutationObserver(checkReady);
    observer.observe(document.body, { childList: true, subtree: true, attributes: true, attributeFilter: ["data-portfolio-pdf-ready", "data-portfolio-smart-phase"] });
    const failureTimer = window.setTimeout(() => {
      if (!ready && !document.querySelector(READY_SELECTOR)) setFailed(true);
    }, FAILURE_TIMEOUT_MS);

    return () => {
      window.clearInterval(progressTimer);
      window.clearTimeout(failureTimer);
      if (completionTimer !== null) window.clearTimeout(completionTimer);
      observer.disconnect();
    };
  }, [active, session]);

  const modal: PrintExportModal | null = !active || dismissed ? null : failed ? {
    status: "error",
    title: "تعذر تجهيز المعاينة",
    message: "استغرق تجهيز ملف الإنجاز وقتًا أطول من المتوقع. يمكنك إغلاق هذه الرسالة والمحاولة مرة أخرى.",
  } : {
    status: "loading",
    title: "جاري تحضير المعاينة",
    message: "يتم الآن تجهيز صفحات ملف الإنجاز، الرجاء الانتظار...",
    progress: ready ? 100 : progress,
  };

  return <>
    {children}
    {modal ? <PrintExportPopCard align="center" modal={modal} onClose={() => setDismissed(true)} onOpenFallback={() => undefined} /> : null}
  </>;
}

function parseRetryAfter(value: string | null) {
  if (!value) return null;
  const seconds = Number(value.trim());
  if (Number.isFinite(seconds) && seconds >= 0) return Math.min(seconds * 1000, 60 * 60 * 1000);
  const timestamp = Date.parse(value);
  if (!Number.isFinite(timestamp)) return null;
  return Math.min(Math.max(0, timestamp - Date.now()), 60 * 60 * 1000);
}

function cloudflareLog(stage: string, details: Record<string, unknown>) {
  console.info(`PORTFOLIO_CLOUDFLARE ${JSON.stringify({ stage, ...details })}`);
}

export async function generatePdfFromUrlWithCloudflare({
  url,
  request,
  waitForSelector = ".pdf-report-page",
  gotoWaitUntil = "networkidle2",
  waitForSelectorTimeoutMs = CLOUDFLARE_BROWSER_OPERATION_TIMEOUT_MS,
  debugLabel,
  landscape = false,
}: {
  url: string;
  request?: Request;
  waitForSelector?: string;
  gotoWaitUntil?: "load" | "domcontentloaded" | "networkidle0" | "networkidle2";
  waitForSelectorTimeoutMs?: number;
  debugLabel?: "portfolio";
  landscape?: boolean;
}): Promise<Uint8Array> {
  const startedAt = Date.now();
  const accountId = process.env.CLOUDFLARE_ACCOUNT_ID?.trim();
  const apiToken = process.env.CLOUDFLARE_BROWSER_RUN_API_TOKEN?.trim();

  if (!accountId || !apiToken) {
    throw new Error("Cloudflare Browser Run PDF credentials are not configured.");
  }

  const controller = new AbortController();
  const timeoutId = setTimeout(
    () => controller.abort(),
    CLOUDFLARE_PDF_TIMEOUT_MS,
  );

  const sessionCookie = request?.headers.get("cookie")
    ?.split(";")
    .map((part) => part.trim())
    .find((part) => part.startsWith("student_guidance_session="));
  const cookies = sessionCookie
    ? [{
        name: "student_guidance_session",
        value: sessionCookie.slice("student_guidance_session=".length),
        url,
        path: "/",
        httpOnly: true,
        secure: url.startsWith("https://"),
        sameSite: "Lax" as const,
      }]
    : undefined;

  const requestBody = {
    url,
    ...(cookies ? { cookies } : {}),
    gotoOptions: {
      waitUntil: gotoWaitUntil,
      timeout: CLOUDFLARE_BROWSER_OPERATION_TIMEOUT_MS,
    },
    waitForSelector: {
      selector: waitForSelector,
      timeout: waitForSelectorTimeoutMs,
      visible: true,
    },
    pdfOptions: {
      format: "a4",
      landscape,
      printBackground: true,
      preferCSSPageSize: true,
      displayHeaderFooter: false,
    },
  };

  if (debugLabel === "portfolio") {
    cloudflareLog("fetch-start", {
      elapsedMs: 0,
      cloudflareTimeoutMs: CLOUDFLARE_PDF_TIMEOUT_MS,
      gotoWaitUntil,
      waitForSelector,
      waitForSelectorTimeoutMs,
    });
  }

  try {
    const response = await fetch(
      `https://api.cloudflare.com/client/v4/accounts/${encodeURIComponent(accountId)}/browser-rendering/pdf`,
      {
        method: "POST",
        headers: {
          Authorization: `Bearer ${apiToken}`,
          "Content-Type": "application/json",
        },
        body: JSON.stringify(requestBody),
        cache: "no-store",
        signal: controller.signal,
      },
    );

    if (debugLabel === "portfolio") {
      cloudflareLog("fetch-response", {
        elapsedMs: Date.now() - startedAt,
        status: response.status,
        ok: response.ok,
      });
    }

    if (!response.ok) {
      const errorBody = (await response.text()).slice(0, 500);
      throw new CloudflareBrowserRunError(response.status, errorBody, parseRetryAfter(response.headers.get("retry-after")));
    }

    const pdfBytes = new Uint8Array(await response.arrayBuffer());

    if (debugLabel === "portfolio") {
      cloudflareLog("pdf-bytes-ready", {
        elapsedMs: Date.now() - startedAt,
        bytesLength: pdfBytes.length,
      });
    }

    const signature = new TextDecoder("ascii").decode(pdfBytes.slice(0, 5));

    if (!pdfBytes.length || signature !== PDF_SIGNATURE) {
      throw new Error("Cloudflare Browser Run returned an invalid PDF body.");
    }

    return pdfBytes;
  } catch (error) {
    if (debugLabel === "portfolio") {
      cloudflareLog("fetch-error", {
        elapsedMs: Date.now() - startedAt,
        errorName: error instanceof Error ? error.name : "UnknownError",
        abortedByLocalController: controller.signal.aborted,
      });
    }

    throw error;
  } finally {
    if (debugLabel === "portfolio") {
      cloudflareLog("fetch-finished", {
        elapsedMs: Date.now() - startedAt,
      });
    }

    clearTimeout(timeoutId);
  }
}

const LOCAL_IMAGE_PREFIXES = ["/uploads/portfolio/", "/uploads/evidence/"] as const;

function isLocalImageUrl(value: string) {
  const pathname = value.split(/[?#]/, 1)[0] || "";
  return LOCAL_IMAGE_PREFIXES.some((prefix) => pathname.startsWith(prefix)) && /\.(?:jpe?g|png|webp)$/i.test(pathname);
}

export function collectPortfolioExportImageUrls(document: PortfolioPrintData) {
  const urls = new Set<string>();
  const visit = (value: unknown) => {
    if (typeof value === "string") {
      if (isLocalImageUrl(value)) urls.add(value);
      return;
    }
    if (Array.isArray(value)) {
      value.forEach(visit);
      return;
    }
    if (value && typeof value === "object") Object.values(value).forEach(visit);
  };
  visit(document);
  return urls;
}

export async function assertPortfolioExportImageSourcesAvailable(document: PortfolioPrintData) {
  const sources = collectPortfolioExportImageUrls(document);
  for (const source of sources) {
    if (source.startsWith("/uploads/portfolio/")) {
      const parts = source.slice("/uploads/portfolio/".length).split("/");
      if (parts.length !== 2 || !isSafeStorageId(parts[0]) || !isSafeUploadedFileName(parts[1]) || !(await storageFileExists(["portfolio", parts[0], parts[1]]))) throw new Error("PORTFOLIO_EXPORT_IMAGE_MISSING");
      continue;
    }
    const fileName = getEvidenceStoredFileNameFromUrl(source);
    if (!fileName || !(await storageFileExists(["evidence", fileName]))) throw new Error("PORTFOLIO_EXPORT_IMAGE_MISSING");
  }
}

export function buildPortfolioExportImageUrl(token: string, source: string) {
  return `/api/portfolio-export/snapshot-image?token=${encodeURIComponent(token)}&source=${encodeURIComponent(source)}`;

export function preparePortfolioDocumentForPdf(document: PortfolioPrintData, token: string) {
  const replace = (value: unknown): unknown => {
    if (typeof value === "string") return isLocalImageUrl(value) ? buildPortfolioExportImageUrl(token, value) : value;
    if (Array.isArray(value)) return value.map(replace);
    if (value && typeof value === "object") return Object.fromEntries(Object.entries(value).map(([key, item]) => [key, replace(item)]));
    return value;
  };
  return replace(document) as PortfolioPrintData;
}

function storageDirectory() {
  return path.join(process.cwd(), ".tmp", "portfolio-export");
}

function storagePath(token: string) {
  if (!TOKEN_PATTERN.test(token)) throw new Error("Invalid portfolio export token.");
  return path.join(storageDirectory(), `${token}.json`);
}

async function cleanupExpiredExports() {
  const directory = storageDirectory();
  try {
    const entries = await fs.readdir(directory, { withFileTypes: true });
    const now = Date.now();
    await Promise.all(entries.filter((entry) => entry.isFile() && TOKEN_PATTERN.test(path.parse(entry.name).name)).map(async (entry) => {
      const filePath = path.join(directory, entry.name);
      try {
        if (now - (await fs.stat(filePath)).mtimeMs > PORTFOLIO_EXPORT_TTL_MS) await fs.unlink(filePath);
      } catch {
        // The file may have been removed by another cleanup.
      }
    }));
  } catch {
    // The directory is created on the first export.
  }
}

export async function createPortfolioExportToken(document: PortfolioPrintData) {
  const token = randomUUID();
  await fs.mkdir(storageDirectory(), { recursive: true });
  await fs.writeFile(storagePath(token), JSON.stringify({ createdAt: Date.now(), document }), { encoding: "utf8", flag: "wx" });
  void cleanupExpiredExports();
  return token;
}

export async function readPortfolioExportToken(token: string) {
  try {
    const filePath = storagePath(token);
    const payload = JSON.parse(await fs.readFile(filePath, "utf8")) as Partial<PortfolioExportPayload>;
    if (typeof payload.createdAt !== "number" || Date.now() - payload.createdAt > PORTFOLIO_EXPORT_TTL_MS || !payload.document || typeof payload.document !== "object") {
      await fs.unlink(filePath).catch(() => undefined);
      return null;
    }
    return payload as PortfolioExportPayload;
  } catch {
    return null;
  }
}

function isPrivateOrLocalHostname(hostname: string) {
  return (
    hostname === "localhost" ||
    hostname === "127.0.0.1" ||
    hostname === "::1" ||
    hostname.endsWith(".local") ||
    /^10\./.test(hostname) ||
    /^192\.168\./.test(hostname) ||
    /^172\.(?:1[6-9]|2\d|3[0-1])\./.test(hostname) ||
    hostname.startsWith("169.254.")
  );
}

export function getPortfolioPublicOrigin() {
  const configured =
    process.env.APP_URL?.trim() ||
    process.env.NEXT_PUBLIC_APP_URL?.trim() ||
    process.env.SITE_URL?.trim();
  const raw = configured || (process.env.NODE_ENV === "production" ? DEFAULT_PRODUCTION_ORIGIN : "");

  if (!raw) throw new Error("PORTFOLIO_PUBLIC_ORIGIN_NOT_CONFIGURED");

  let origin: URL;
  try {
    origin = new URL(raw);
  } catch {
    throw new Error("PORTFOLIO_PUBLIC_ORIGIN_INVALID");
  }

  const hostname = origin.hostname.toLowerCase();
  const productionHostAllowed =
    process.env.NODE_ENV !== "production" ||
    hostname === "teachix.sa" ||
    hostname === "www.teachix.sa";

  if (
    origin.protocol !== "https:" ||
    origin.username ||
    origin.password ||
    isPrivateOrLocalHostname(hostname) ||
    !productionHostAllowed
  ) {
    throw new Error("PORTFOLIO_PUBLIC_ORIGIN_NOT_REACHABLE");
  }

  return origin.origin;
}

  | "lock-connect"
  | "get-lock"
  | "recover-expired"

function sanitizeError(error: unknown) {
  const message = error instanceof Error ? error.message : String(error);
  return message
    .replace(/https?:\/\/[^\s"'<>]+/gi, "[redacted-url]")
    .replace(/token=[^&\s]+/gi, "token=[redacted]")
    .replace(/[\r\n]+/g, " ")
    .slice(0, 1000);
}

function errorProperty(error: unknown, property: string) {
  return error && typeof error === "object" && property in error
    ? (error as Record<string, unknown>)[property]
    : undefined;
}

function isRateLimitError(error: unknown) {
  return errorProperty(error, "status") === 429 || /\b429\b|rate limit/i.test(error instanceof Error ? error.message : String(error));
}

function nextRetryAt(attemptCount: number, error?: unknown) {
  const rateLimit = isRateLimitError(error);
  const base = rateLimit ? RATE_LIMIT_RETRY_BASE_MS : RETRY_BASE_MS;
  const exponentialDelay = base * 2 ** Math.max(0, attemptCount - 1);
  const retryAfterMs = Number(errorProperty(error, "retryAfterMs"));
  const requestedDelay = Number.isFinite(retryAfterMs) && retryAfterMs > 0
    ? retryAfterMs
    : exponentialDelay;
  const jitterMs = Math.floor(Math.random() * Math.max(1000, Math.min(base, 30 * 1000)));
  return new Date(Date.now() + Math.min(Math.max(requestedDelay, exponentialDelay) + jitterMs, RETRY_MAX_MS));
}

function ensureDatabaseUrlSearchParam(url: URL, key: string, value: string) {
  if (!url.searchParams.has(key)) {
    url.searchParams.set(key, value);
  }
}

function portfolioSnapshotPdfLockConnectionUrl(databaseUrl: string) {
  const url = new URL(databaseUrl);

  // PrismaMariaDb rewrites mysql:// to mariadb:// before passing the URL
  // to the MariaDB connector. The dedicated advisory-lock connection must
  // use the same compatible protocol and connection options.
  if (url.protocol === "mysql:") {
    url.protocol = "mariadb:";
  } else if (url.protocol !== "mariadb:") {
    throw new Error("UNSUPPORTED_DATABASE_PROTOCOL");
  }

  ensureDatabaseUrlSearchParam(url, "acquireTimeout", process.env.PRISMA_POOL_TIMEOUT_MS || "30000");
  ensureDatabaseUrlSearchParam(url, "connectTimeout", process.env.PRISMA_CONNECT_TIMEOUT_MS || "15000");
  ensureDatabaseUrlSearchParam(url, "allowPublicKeyRetrieval", "true");
  ensureDatabaseUrlSearchParam(url, "prepareCacheLength", "0");

  return url.toString();
}

function annotateClaimError(error: unknown, claimOperation: PortfolioSnapshotPdfClaimOperation) {
  if (error instanceof Error) {
    Object.defineProperty(error, "claimOperation", {
      configurable: true,
      value: claimOperation,
    });
    return error;
  }

  const wrapped = new Error("Portfolio snapshot PDF claim failed");
  Object.defineProperty(wrapped, "claimOperation", {
    configurable: true,
    value: claimOperation,
  });
  return wrapped;
}

async function acquirePortfolioSnapshotPdfGlobalLock(): Promise<PortfolioSnapshotPdfGlobalLock | null> {
  const databaseUrl = process.env.DATABASE_URL;
  if (!databaseUrl) throw new Error("DATABASE_URL is not defined");

  let claimOperation: PortfolioSnapshotPdfClaimOperation = "lock-connect";
  let connection: mariadb.Connection | undefined;
  try {
    connection = await mariadb.createConnection(portfolioSnapshotPdfLockConnectionUrl(databaseUrl));
  } catch (error) {
    throw annotateClaimError(error, claimOperation);
  }

  let acquired = false;
  try {
    claimOperation = "get-lock";
    const rows = await connection.query("SELECT GET_LOCK(?, 0) AS acquired", [GLOBAL_LOCK_NAME]);
    acquired = Number(rows[0]?.acquired) === 1;
    if (!acquired) {
      await connection.end();
      return null;
    }

    let released = false;
    return {
      release: async () => {
        if (released) return;
        released = true;
        try {
          await connection.query("SELECT RELEASE_LOCK(?)", [GLOBAL_LOCK_NAME]);
        } finally {
          await connection.end();
        }
      },
    };
  } catch (error) {
    if (acquired) await connection.query("SELECT RELEASE_LOCK(?)", [GLOBAL_LOCK_NAME]).catch(() => undefined);
    await connection.end().catch(() => undefined);
    throw annotateClaimError(error, claimOperation);
  }
}

export async function registerPortfolioSnapshotPdf(snapshotId: string, document: unknown, options?: { forceRetry?: boolean }) {

  const existing = await prisma.portfolioSnapshotPdfArtifact.findUnique({

      byteSize: null,
      nextAttemptAt: null,
      leaseExpiresAt: null,
      claimToken: null,
      lastError: null,
      generatedAt: null,
    },
    create: { snapshotId, contentHash },
    select: { id: true, contentHash: true, status: true },
  });
}

export async function claimNextPortfolioSnapshotPdfJob() {
  let claimOperation: PortfolioSnapshotPdfClaimOperation = "lock-connect";
  let globalLock: PortfolioSnapshotPdfGlobalLock | null;
  try {
    globalLock = await acquirePortfolioSnapshotPdfGlobalLock();
  } catch (error) {
    throw annotateClaimError(
      error,
      (error as PortfolioSnapshotPdfClaimError).claimOperation ?? claimOperation,
    );
  }
  if (!globalLock) return { kind: "GLOBAL_BUSY" as const };

  const releaseOnClaimFailure = async () => globalLock.release().catch(() => undefined);
  try {
    claimOperation = "recover-expired";
    await recoverExpiredPortfolioSnapshotPdfJobs();
    const now = new Date();
    claimOperation = "find-first";
    const candidate = await prisma.portfolioSnapshotPdfArtifact.findFirst({
      where: {
        status: "PENDING",
        OR: [{ nextAttemptAt: null }, { nextAttemptAt: { lte: now } }],
      },
      orderBy: { createdAt: "asc" },
    });
    if (!candidate) {
      await releaseOnClaimFailure();
      return { kind: "NO_ELIGIBLE_JOB" as const };
    }

    const claimToken = randomUUID();
    claimOperation = "claim-update";
    const claimed = await prisma.portfolioSnapshotPdfArtifact.updateMany({
      where: { id: candidate.id, status: "PENDING", contentHash: candidate.contentHash },
      data: {
        status: "PROCESSING",
        claimToken,
        leaseExpiresAt: new Date(Date.now() + LEASE_MS),
        attemptCount: { increment: 1 },
        lastError: null,
      },
    });
    if (claimed.count !== 1) {
      await releaseOnClaimFailure();
      return { kind: "CLAIM_CONFLICT" as const };
    }

    claimOperation = "load-job";
    const job = await prisma.portfolioSnapshotPdfArtifact.findUnique({
      where: { id: candidate.id },
      include: { snapshot: { select: { id: true, snapshotJson: true } } },
    });
    if (!job) {
      await releaseOnClaimFailure();
      return { kind: "CLAIM_CONFLICT" as const };
    }
    return { kind: "CLAIMED" as const, job, releaseGlobalLock: globalLock.release };
  } catch (error) {
    await releaseOnClaimFailure();
    throw annotateClaimError(
      error,
      (error as PortfolioSnapshotPdfClaimError).claimOperation ?? claimOperation,
    );
  }
}

export async function failPortfolioSnapshotPdfJob(input: { id: string; claimToken: string; error: unknown }) {
  const current = await prisma.portfolioSnapshotPdfArtifact.findFirst({
    where: { id: input.id, status: "PROCESSING", claimToken: input.claimToken },
    select: { attemptCount: true },
  });
  if (!current) return { count: 0, retryable: false };

  const retryable = current.attemptCount < MAX_ATTEMPTS;
  const result = await prisma.portfolioSnapshotPdfArtifact.updateMany({
    where: { id: input.id, status: "PROCESSING", claimToken: input.claimToken },
    data: retryable
      ? {
          status: "PENDING",
          nextAttemptAt: nextRetryAt(current.attemptCount, input.error),
          leaseExpiresAt: null,
          claimToken: null,
          lastError: sanitizeError(input.error),
        }
      : {
          status: "FAILED",
          nextAttemptAt: null,
          leaseExpiresAt: null,
          claimToken: null,
          lastError: sanitizeError(input.error),
        },
  });
  return { count: result.count, retryable };
}

export async function createPortfolioSnapshot(
  user: SnapshotUser,
  portfolioId: string,
  input: { name?: string; notes?: string },
) {
  await requireOwnedPersonalPortfolio(user, portfolioId);
  assertPortfolioActor(user);
  const workspace = await getPortfolioWorkspace(user, portfolioId);
  if (!workspace.ok) {
    throw new PortfolioServiceError(400, "تعذر تجهيز ملف الإنجاز للاعتماد.");
  }

  const capturedAt = new Date();
  const document = toSnapshotDocument(workspace, user, capturedAt);
  const summary = toSummary(document);

  const snapshot = await prisma.portfolioSnapshot.create({
    data: {
      portfolioId,
      ownerUserId: user.id,
      schoolAccountId: user.schoolAccountId!,
      name: input.name?.trim() || defaultSnapshotName(capturedAt),
      notes: input.notes?.trim() || null,
      roleAtCreation: user.role,
      snapshotVersion: PORTFOLIO_SNAPSHOT_VERSION,
      summaryJson: asJson(summary),
      snapshotJson: asJson(document),
    },
    select: {
      id: true,
      name: true,
      notes: true,
      roleAtCreation: true,
      snapshotVersion: true,
      summaryJson: true,
      createdAt: true,
    },
  });

  let pdfPreparation: "PENDING" | "REGISTRATION_FAILED" = "PENDING";
  try {
    await registerPortfolioSnapshotPdf(snapshot.id, document);
  } catch (error) {
    pdfPreparation = "REGISTRATION_FAILED";
    console.error("PORTFOLIO_SNAPSHOT_PDF_REGISTRATION_FAILED", {
      snapshotId: snapshot.id,
      error: error instanceof Error ? error.message : String(error),
    });
  }

  return { ...snapshot, pdfPreparation };
}

export async function listPortfolioSnapshots(user: PortfolioActor, portfolioId: string) {
  await requireOwnedPortfolio(user, portfolioId, { historicalPersonalRead: true });
  assertPortfolioActor(user);
  const snapshots = await prisma.portfolioSnapshot.findMany({
    where: {
      portfolioId,
      ownerUserId: user.id,
    },
    orderBy: { createdAt: "desc" },
    select: {
      id: true,
      name: true,
      notes: true,
      roleAtCreation: true,
      snapshotVersion: true,
      summaryJson: true,
      createdAt: true,
      pdfArtifact: { select: { status: true, byteSize: true, generatedAt: true } },
    },
  });

  return snapshots.map((snapshot) => ({
    id: snapshot.id,
    name: snapshot.name,
    notes: snapshot.notes || "",
    roleAtCreation: snapshot.roleAtCreation,
    snapshotVersion: snapshot.snapshotVersion,
    createdAt: snapshot.createdAt.toISOString(),
    summary: readPortfolioSnapshotSummary(snapshot.summaryJson),
    // A missing artifact is retryable (older snapshots and registrations that failed after save).
    pdfPreparationStatus: snapshot.pdfArtifact?.status || "FAILED",
    pdfByteSize: snapshot.pdfArtifact?.byteSize || null,
  }));
}

export async function getPortfolioSnapshot(user: PortfolioActor, snapshotId: string) {
  assertPortfolioActor(user);
  const snapshot = await prisma.portfolioSnapshot.findFirst({
    where: {
      id: snapshotId,
      ownerUserId: user.id,
    },
  });
  if (!snapshot) {
    throw new PortfolioServiceError(404, "نسخة ملف الإنجاز غير موجودة أو لا تملك صلاحية الوصول إليها.");
  }

  await requireOwnedPortfolio(user, snapshot.portfolioId, { historicalPersonalRead: true });
  const document = parsePortfolioSnapshotDocument(snapshot.snapshotJson);
  return {
    ...snapshot,
    document: {
      ...document,
      showWeights: document.showWeights ?? snapshot.roleAtCreation === "TEACHER",
    },
  };
}

export async function deletePortfolioSnapshot(user: PortfolioActor, portfolioId: string, snapshotId: string) {
  assertPortfolioActor(user);
  const snapshot = await prisma.portfolioSnapshot.findFirst({
    where: {
      id: snapshotId,
      portfolioId,
      ownerUserId: user.id,
    },
    select: {
      id: true,
      portfolioId: true,
    },
  });

  if (!snapshot) {
    throw new PortfolioServiceError(404, "نسخة ملف الإنجاز غير موجودة أو لا تملك صلاحية حذفها.");
  }

  await requireOwnedPortfolio(user, snapshot.portfolioId, { historicalPersonalRead: true });
  await prisma.portfolioSnapshot.delete({ where: { id: snapshot.id } });
}

import type { PortfolioPrintData } from "@/components/portfolio/print/portfolio-print-types";
import { z } from "zod";

export const PORTFOLIO_SNAPSHOT_VERSION = 1 as const;

export const portfolioSnapshotCreateSchema = z.object({

export type PortfolioSnapshotDocumentV1 = PortfolioPrintData & {

function isRecord(value: unknown): value is Record<string, unknown> {
  return Boolean(value) && typeof value === "object" && !Array.isArray(value);
}

export function parsePortfolioSnapshotDocument(value: unknown): PortfolioSnapshotDocumentV1 {
  if (!isRecord(value) || value.snapshotVersion !== PORTFOLIO_SNAPSHOT_VERSION) {
    throw new Error("إصدار نسخة ملف الإنجاز غير مدعوم.");
  }
  if (
    !isRecord(value.portfolio) ||
    !isRecord(value.owner) ||
    !isRecord(value.school) ||
    !Array.isArray(value.sections) ||
    !Array.isArray(value.qualificationItems) ||
    !Array.isArray(value.performanceSections) ||
    !Array.isArray(value.customEvidence) ||
    !isRecord(value.biography) ||
    !isRecord(value.educationIdentity) ||
    !isRecord(value.totals)
  ) {
    throw new Error("بيانات نسخة ملف الإنجاز غير مكتملة.");
  }

  const performanceSections = value.performanceSections.map((section) => {
    if (!isRecord(section)) return section;
    return { ...section, linkedOutputs: Array.isArray(section.linkedOutputs) ? section.linkedOutputs : [] };
  });
  return { ...value, performanceSections } as PortfolioSnapshotDocumentV1;
}

export function readPortfolioSnapshotSummary(value: unknown): PortfolioSnapshotSummary | null {
  if (!isRecord(value)) return null;
  return {
    title: typeof value.title === "string" ? value.title : "",
    academicYear: typeof value.academicYear === "string" ? value.academicYear : "",
    term: typeof value.term === "string" ? value.term : "",
    themeId: typeof value.themeId === "string" ? value.themeId : "",
    ownerName: typeof value.ownerName === "string" ? value.ownerName : "",
    reportCount: typeof value.reportCount === "number" ? value.reportCount : 0,
    evidenceCount: typeof value.evidenceCount === "number" ? value.evidenceCount : 0,
  };
}
