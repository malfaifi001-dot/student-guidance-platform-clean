MODEL: TimetableProject
============================================================
model TimetableProject {
  id String @id @default(cuid())

  schoolAccountId String
  schoolAccount   SchoolAccount @relation(fields: [schoolAccountId], references: [id], onDelete: Cascade)

  createdById String
  createdBy   User   @relation("TimetableProjectCreator", fields: [createdById], references: [id], onDelete: Restrict)

  name         String
  academicYear String
  semester     String

  status TimetableProjectStatus @default(DRAFT)

  daysJson     Json
  periodsJson  Json
  settingsJson Json?

  teachers          TimetableTeacher[]
  classes           TimetableClass[]
  subjects          TimetableSubject[]
  classSubjects     TimetableClassSubject[]
  assignments       TimetableAssignment[]
  waitingPolicy     TimetableWaitingPolicy?
  dailyAbsences     TimetableDailyAbsence[]
  substitutions     TimetableSubstitution[]
  supervisionDuties TimetableSupervisionDuty[]

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  @@index([schoolAccountId])
  @@index([createdById])
  @@index([status])
  @@index([createdAt])
}

MODEL: TimetableTeacher
============================================================
model TimetableTeacher {
  id String @id @default(cuid())

  projectId String
  project   TimetableProject @relation(fields: [projectId], references: [id], onDelete: Cascade)

  userId String?
  user   User?   @relation(fields: [userId], references: [id], onDelete: SetNull)

  name          String
  specialty     String?
  maxWeeklyLoad Int     @default(24)
  isActive      Boolean @default(true)

  unavailableSlotsJson Json?

  assignments            TimetableAssignment[]
  dailyAbsences          TimetableDailyAbsence[]
  originalSubstitutions  TimetableSubstitution[]          @relation("OriginalSubstitutionTeacher")
  assignedSubstitutions  TimetableSubstitution[]          @relation("AssignedSubstitutionTeacher")
  supervisionAssignments TimetableSupervisionAssignment[]

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  @@unique([projectId, userId])
  @@index([projectId])
  @@index([userId])
  @@index([isActive])
}

MODEL: TimetableAssignment
============================================================
model TimetableAssignment {
  id String @id @default(cuid())

  projectId String
  project   TimetableProject @relation(fields: [projectId], references: [id], onDelete: Cascade)

  teacherId String
  teacher   TimetableTeacher @relation(fields: [teacherId], references: [id], onDelete: Cascade)

  classId String
  class   TimetableClass @relation(fields: [classId], references: [id], onDelete: Cascade)

  subjectId String
  subject   TimetableSubject @relation(fields: [subjectId], references: [id], onDelete: Cascade)

  assignedLessons Int
  singlePeriods   Int @default(0)
  doublePeriods   Int @default(0)

  fixedSlotsJson Json?

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  @@unique([projectId, teacherId, classId, subjectId])
  @@index([projectId])
  @@index([teacherId])
  @@index([classId])
  @@index([subjectId])
}
