import { NextResponse } from "next/server";

import { prisma } from "@/lib/prisma";

type RouteContext = {
  params: Promise<{
    serviceSlug: string;
  }>;
};

export async function POST(
  request: Request,
  context: RouteContext
) {
  try {
    const { serviceSlug } = await context.params;

    const draftWorkflow = await prisma.workflow.findFirst({
      where: {
        service: {
          slug: serviceSlug,
        },
        status: "DRAFT",
      },
      orderBy: {
        version: "desc",
      },
    });

    if (!draftWorkflow) {
      return NextResponse.json(
        {
          error: "?? ???? Draft Workflow.",
        },
        {
          status: 404,
        }
      );
    }

    await prisma.workflow.updateMany({
      where: {
        serviceId: draftWorkflow.serviceId,
        status: "PUBLISHED",
      },
      data: {
        status: "ARCHIVED",
        isActive: false,
      },
    });

    const publishedWorkflow = await prisma.workflow.update({
      where: {
        id: draftWorkflow.id,
      },
      data: {
        status: "PUBLISHED",
        isActive: true,
        publishedAt: new Date(),
      },
    });

    return NextResponse.json({
      success: true,
      workflowId: publishedWorkflow.id,
    });
  } catch (error) {
    return NextResponse.json(
      {
        error:
          error instanceof Error
            ? error.message
            : "??? ??? ????? ??? Workflow.",
      },
      {
        status: 400,
      }
    );
  }
}

  async function publishWorkflow() {
    try {
      setLoading(true);

      const response = await fetch(
        `/api/dashboard/admin/workflows/${serviceSlug}/publish`,
        {

      );

      const data = await response.json();
